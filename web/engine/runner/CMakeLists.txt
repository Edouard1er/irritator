cmake_minimum_required(VERSION 3.20)
project(irritator_web_runner VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(IRRITATOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(IRRITATOR_BUILD_DIR "${IRRITATOR_ROOT}/build-wrapper-test3" CACHE PATH "Matching engine build directory")
unset(IRRITATOR_LIBRARY CACHE)
find_library(IRRITATOR_LIBRARY NAMES irritators-0.1 PATHS "${IRRITATOR_BUILD_DIR}/lib" "${IRRITATOR_BUILD_DIR}/lib/${CMAKE_BUILD_TYPE}" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
add_library(irritator-native-wrapper STATIC native_wrapper.cpp)
target_include_directories(irritator-native-wrapper PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${IRRITATOR_ROOT}/lib/include")
target_link_libraries(irritator-native-wrapper PUBLIC "${IRRITATOR_LIBRARY}" Threads::Threads)
target_compile_definitions(irritator-native-wrapper PUBLIC $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
if(MSVC)
  target_compile_options(irritator-native-wrapper PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030)
endif()
set(CONTRACTS "${CMAKE_CURRENT_LIST_DIR}/../../packages/contracts")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CONTRACTS}/runner-capabilities.json")
file(READ "${CONTRACTS}/runner-capabilities.json" RUNNER_CAPABILITIES_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CONTRACTS}/catalog.json" "${CONTRACTS}/request.schema.json" "${CONTRACTS}/request-v2.schema.json" "${CONTRACTS}/request-v3.schema.json")
file(READ "${CONTRACTS}/catalog.json" CATALOG_JSON)
file(READ "${CONTRACTS}/request.schema.json" REQUEST_SCHEMA_JSON)
file(READ "${CONTRACTS}/request-v2.schema.json" REQUEST_V2_SCHEMA_JSON)
file(READ "${CONTRACTS}/request-v3.schema.json" REQUEST_V3_SCHEMA_JSON)
configure_file(contracts.hpp.in contracts.hpp @ONLY)
add_executable(irritator-runner main.cpp)
target_include_directories(irritator-runner PRIVATE "${IRRITATOR_ROOT}/lib/include" "${IRRITATOR_ROOT}/external/rapidjson/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(irritator-runner PRIVATE irritator-native-wrapper)
target_compile_definitions(irritator-runner PRIVATE $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
if(MSVC)
  target_compile_options(irritator-runner PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030)
endif()
enable_testing()
find_program(NODE_EXECUTABLE node REQUIRED)
add_test(NAME runner-contract COMMAND "${CMAKE_COMMAND}" -E env "IRRITATOR_RUNNER=$<TARGET_FILE:irritator-runner>" "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_LIST_DIR}/tests.mjs")
set_tests_properties(runner-contract PROPERTIES TIMEOUT 60)
foreach(suite hierarchy qss-catalog)
  add_test(NAME runner-${suite} COMMAND "${CMAKE_COMMAND}" -E env "IRRITATOR_RUNNER=$<TARGET_FILE:irritator-runner>" "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_LIST_DIR}/${suite}-tests.mjs")
  set_tests_properties(runner-${suite} PROPERTIES TIMEOUT 120)
endforeach()
add_test(NAME runner-contracts-build COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../node_modules/typescript/bin/tsc" -p "${CMAKE_CURRENT_LIST_DIR}/../../apps/api/tsconfig.json")
set_tests_properties(runner-contracts-build PROPERTIES FIXTURES_SETUP contracts-v3 TIMEOUT 60)
add_test(NAME runner-wrapper-v3 COMMAND "${CMAKE_COMMAND}" -E env "IRRITATOR_RUNNER=$<TARGET_FILE:irritator-runner>" "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_LIST_DIR}/wrapper-v3-tests.mjs")
set_tests_properties(runner-wrapper-v3 PROPERTIES FIXTURES_REQUIRED contracts-v3 TIMEOUT 120)
add_subdirectory(tests-wrapper)

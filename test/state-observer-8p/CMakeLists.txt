cmake_minimum_required(VERSION 3.20)
project(state_observer_8p_proof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(IRRITATOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(IRRITATOR_BUILD_DIR "" CACHE PATH "Matching engine build directory")
find_library(IRRITATOR_LIBRARY NAMES irritators-0.1
  PATHS "${IRRITATOR_BUILD_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
add_executable(state-observer-8p-proof scheduler-proof.cpp)
add_executable(state-observer-8p-arbitration candidate-arbitration.cpp)
add_executable(state-observer-8p-snapshot snapshot-proof.cpp)
foreach(target state-observer-8p-proof state-observer-8p-arbitration state-observer-8p-snapshot)
  target_include_directories(${target} PRIVATE "${IRRITATOR_ROOT}/lib/include")
  target_link_libraries(${target} PRIVATE "${IRRITATOR_LIBRARY}" Threads::Threads)
  target_compile_definitions(${target} PRIVATE $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030 /fp:precise)
  endif()
endforeach()
enable_testing()
add_test(NAME closed-cohort-command-arbitration COMMAND state-observer-8p-arbitration)
add_test(NAME reproduces-inflight-snapshot-obstruction COMMAND state-observer-8p-snapshot)
add_test(NAME requires-exact-inflight-snapshot COMMAND state-observer-8p-snapshot --require-exact)
add_test(NAME reproduces-scheduler-obstruction COMMAND state-observer-8p-proof)
# This exact-date acceptance fails on the audited engine. Keep it as a real
# requirement, without WILL_FAIL or a declaration of PDEVS conformity.
add_test(NAME requires-exact-physical-deadline COMMAND state-observer-8p-proof --require-exact)

cmake_minimum_required(VERSION 3.20)
project(state_observer_8p_bis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(IRRITATOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(IRRITATOR_BUILD_DIR "" CACHE PATH "Matching engine build directory")
find_library(IRRITATOR_LIBRARY NAMES irritators-0.1 PATHS "${IRRITATOR_BUILD_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
set_property(SOURCE "../state-observer-8p/scheduler-proof.cpp" "../state-observer-8p/snapshot-proof.cpp" "../state-observer-8p/candidate-arbitration.cpp" snapshot-lifecycle.cpp nested-snapshot.cpp differential.cpp
  APPEND PROPERTY OBJECT_DEPENDS "${IRRITATOR_ROOT}/lib/include/irritator/core.hpp" "${IRRITATOR_ROOT}/lib/include/irritator/event.hpp")
foreach(pair "deadline;scheduler-proof.cpp" "snapshot;snapshot-proof.cpp" "arbitration;candidate-arbitration.cpp")
  list(GET pair 0 name)
  list(GET pair 1 source)
  add_executable(bis-${name} "../state-observer-8p/${source}")
  target_include_directories(bis-${name} PRIVATE "${IRRITATOR_ROOT}/lib/include")
  target_link_libraries(bis-${name} PRIVATE "${IRRITATOR_LIBRARY}" Threads::Threads)
  target_compile_definitions(bis-${name} PRIVATE $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
  if(MSVC)
    target_compile_options(bis-${name} PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030 /fp:precise)
  endif()
endforeach()
enable_testing()
add_executable(bis-lifecycle snapshot-lifecycle.cpp)
target_include_directories(bis-lifecycle PRIVATE "${IRRITATOR_ROOT}/lib/include")
target_link_libraries(bis-lifecycle PRIVATE "${IRRITATOR_LIBRARY}" Threads::Threads)
target_compile_definitions(bis-lifecycle PRIVATE $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
if(MSVC)
  target_compile_options(bis-lifecycle PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030 /fp:precise)
endif()
add_test(NAME snapshot-lifecycle COMMAND bis-lifecycle)
add_executable(bis-nested nested-snapshot.cpp)
target_include_directories(bis-nested PRIVATE "${IRRITATOR_ROOT}/lib/include")
target_link_libraries(bis-nested PRIVATE "${IRRITATOR_LIBRARY}" Threads::Threads)
target_compile_definitions(bis-nested PRIVATE $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
if(MSVC)
  target_compile_options(bis-nested PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030 /fp:precise)
endif()
add_test(NAME nested-snapshot COMMAND bis-nested)
foreach(target bis-differential bis-differential-before)
  add_executable(${target} differential.cpp)
  target_include_directories(${target} PRIVATE "${IRRITATOR_ROOT}/lib/include")
  target_compile_definitions(${target} PRIVATE $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030 /fp:precise)
  endif()
endforeach()
target_link_libraries(bis-differential PRIVATE "${IRRITATOR_LIBRARY}" Threads::Threads)
target_include_directories(bis-differential-before BEFORE PRIVATE "${IRRITATOR_ROOT}/build-web-test8p-bis/baseline/include")
target_link_libraries(bis-differential-before PRIVATE "${IRRITATOR_ROOT}/build-web-test8p/${CMAKE_BUILD_TYPE}/engine/lib/irritators-0.1.lib" Threads::Threads)
add_test(NAME exact-historical-and-instrumented-qss-traces
  COMMAND "${CMAKE_COMMAND}" "-DBEFORE=$<TARGET_FILE:bis-differential-before>" "-DAFTER=$<TARGET_FILE:bis-differential>" "-DOUT=${CMAKE_CURRENT_BINARY_DIR}" -P "${CMAKE_CURRENT_LIST_DIR}/compare.cmake")
add_test(NAME requires-exact-inflight-snapshot COMMAND bis-snapshot --require-exact)
add_test(NAME requires-exact-physical-deadline COMMAND bis-deadline --require-exact)
add_test(NAME closed-cohort-command-arbitration COMMAND bis-arbitration)
# Frozen, statically linked 8P binaries retain both historical failure witnesses.
# The seven original diagnostic source files and their CTest definitions remain intact.
add_test(NAME historical-reproduces-inflight-snapshot-obstruction COMMAND "${IRRITATOR_ROOT}/build-web-test8p/${CMAKE_BUILD_TYPE}/proof/state-observer-8p-snapshot.exe")
add_test(NAME historical-reproduces-scheduler-obstruction COMMAND "${IRRITATOR_ROOT}/build-web-test8p/${CMAKE_BUILD_TYPE}/proof/state-observer-8p-proof.exe")
add_executable(bis-protocol protocol-lifecycle.cpp)
target_include_directories(bis-protocol PRIVATE "${IRRITATOR_ROOT}/lib/include")
target_link_libraries(bis-protocol PRIVATE "${IRRITATOR_LIBRARY}" Threads::Threads)
target_compile_definitions(bis-protocol PRIVATE $<$<CONFIG:Debug>:IRRITATOR_ENABLE_DEBUG>)
if(MSVC)
  target_compile_options(bis-protocol PRIVATE /utf-8 /EHsc /bigobj /Zc:__cplusplus /wd5030 /fp:precise)
endif()
set_property(SOURCE protocol-lifecycle.cpp APPEND PROPERTY OBJECT_DEPENDS "${IRRITATOR_ROOT}/lib/include/irritator/core.hpp" "${IRRITATOR_ROOT}/lib/include/irritator/event.hpp" "${IRRITATOR_ROOT}/lib/include/irritator/qss-probe.hpp")
foreach(case cancellation capacity snapshots structure clones)
  add_test(NAME protocol-${case} COMMAND bis-protocol ${case})
endforeach()
get_property(bis_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${bis_tests} PROPERTIES TIMEOUT 60)
